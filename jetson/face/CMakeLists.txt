# beni_face (§3.11). Host build on the Nano (gcc-7.5, C++14): make face
cmake_minimum_required(VERSION 3.10)
project(beni_face CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
  set(CMAKE_CXX_FLAGS_RELEASE "-O2 -mcpu=cortex-a57 -DNDEBUG")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-app-1.0)
pkg_check_modules(ZMQ REQUIRED IMPORTED_TARGET libzmq)
add_executable(beni_face beni_face.cpp)
target_include_directories(beni_face PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../shared)
target_compile_options(beni_face PRIVATE -Wall -Wextra)
target_link_libraries(beni_face PRIVATE PkgConfig::GST PkgConfig::ZMQ)
