cmake_minimum_required(VERSION 3.8)
project(beni_zmq_bridge CXX)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
add_compile_options(-Wall -Wextra -Wno-unused-parameter)

# shared/schemas.hpp: the repo tree when built in place, /ws/shared in the container (-DBENI_SHARED_DIR).
set(BENI_SHARED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../shared" CACHE PATH "repo shared/ directory")
if(NOT EXISTS "${BENI_SHARED_DIR}/schemas.hpp")
  message(FATAL_ERROR "schemas.hpp not found under BENI_SHARED_DIR=${BENI_SHARED_DIR}")
endif()

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(slam_toolbox REQUIRED)
find_library(ZMQ_LIB zmq)
find_path(ZMQ_INCLUDE zmq.h)
if(NOT ZMQ_LIB OR NOT ZMQ_INCLUDE)
  message(FATAL_ERROR "libzmq not found (apt install libzmq3-dev)")
endif()

add_library(zmq_bridge SHARED src/bridge.cpp)
target_include_directories(zmq_bridge PRIVATE "${BENI_SHARED_DIR}" "${ZMQ_INCLUDE}")
target_link_libraries(zmq_bridge "${ZMQ_LIB}")
ament_target_dependencies(zmq_bridge rclcpp rclcpp_action rclcpp_components geometry_msgs nav_msgs nav2_msgs
  sensor_msgs std_msgs std_srvs tf2 tf2_ros tf2_geometry_msgs slam_toolbox)
rclcpp_components_register_node(zmq_bridge PLUGIN "beni::Bridge" EXECUTABLE zmq_bridge_node)

install(TARGETS zmq_bridge ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
ament_package()
